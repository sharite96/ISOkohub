v.label||

function normalizeList(d,keys=[]){

if(Array.isArray(d))return d;

if(!d||typeof d!=="object")return [];

for(const k of keys){

if(Array.isArray(d[k])){
return d[k];
}

}

return [];
}

return "vehicle";
}

}

return isService(item)?"service":"product";
}

headers
});

}catch(e){

throw new Error(
"Cannot connect to IsokoHub API."
);

}

let data={};

try{

const text=await r.text();

if(text){

try{
data=JSON.parse(text);
}catch(e){
data={message:text};
}

}

}catch(e){}

if(!r.ok){

throw new Error(
data.error||
data.message||
`Request failed (${r.status})`
);

}

return data;

}

async function loadMarketplace(params=""){

$("products").innerHTML=
'<div class="empty">🔄 Loading marketplace...</div>';

let query=params;

if(query&&!query.startsWith("?")){
query="?"+query;
}

if(!query){
query="?limit=100";
}

let productResult=[];
let serviceResult=[];

const results=
await Promise.allSettled([

api("/products"+query),
api("/services"+query)

]);

if(results[0].status==="fulfilled"){

productResult=
extractItems(
results[0].value,
"product"
).map(x=>({
...x,
__kind:"product"
}));

}

if(results[1].status==="fulfilled"){

serviceResult=
extractItems(
results[1].value,
"service"
).map(x=>({
...x,
__kind:"service"
}));

}

let combined=[
...productResult,
...serviceResult
];

let seen=new Set();

combined=combined.filter(x=>{

let key=itemKind(x)+":"+String(x.id);

if(
x.id!==undefined &&
x.id!==null &&
seen.has(key)
){
return false;
}

if(
x.id!==undefined &&
x.id!==null
){
seen.add(key);
}

return true;

});

products=combined;

render(products);

}

async function search(

){

$("filterCountry").value=country;
}

$("filterCategory").value=category;
}

let params=new URLSearchParams();

if(q)params.set("search",q);
if(category)params.set("category",category);
if(country)params.set("country",country);

params.set("limit","100");

let endpoint="/products";

if(type==="service")endpoint="/services";

if(
type==="property"||
type==="vehicle"||
type==="job"
){
endpoint="/"+type+"s";
}

try{

if(type){

await loadEndpoint(
endpoint+"?"+params.toString(),
type
);

}else{

await loadMarketplace(
"?"+params.toString()
);

}

$("resultTitle").textContent=
q
?`Search results for "${q}"`
:
category
?category
:"Marketplace";

$("resultSub").textContent=
type
?
type.toUpperCase()+" listings available worldwide."
:
"Products and services available worldwide.";

}catch(e){

$("products").innerHTML=
`<div class="empty">❌ ${esc(e.message)}</div>`;

}

}

async function addById(id,kind=""){

let p=
products.find(x=>
String(x?.id)===String(id)
);

if(!p){

try{
p=await getListingById(id,kind);
}catch(e){
alert(e.message);
return;
}

}

if(itemKind(p)!=="product"){

view(id,itemKind(p));
return;

}

if(
!cartData.some(
x=>String(x?.id)===String(id)
)
){

cartData.push({
...p,
__kind:"product"
});

updateCounts();

}

cartOpen();

}

function cartOpen(){

$("cartItems").innerHTML="";

if(!cartData.length){

$("cartItems").innerHTML=
`
<div class="empty">
🛒 Your cart is empty.
</div>
`;

$("cartTotal").textContent="";

openM("cartModal");

return;

}

let total=0;

$("cartItems").innerHTML=
cartData.map((p,i)=>{

total+=Number(p?.price||0);

let img=
p?.image_url||
p?.image||
"";

return `
<div class="cart-row">

${
img
?
`
<img
class="cart-img"
src="${esc(img)}"
alt="${esc(p?.title||"Product")}"
>
`
:
`
<div class="cart-img">📦</div>
`
}

<div>

<strong>
${esc(p?.title||p?.name||"")}
</strong>

<div class="meta">
${money(
p?.price,
p?.currency||"RWF"
)}
</div>

</div>

<button
class="btn red sm"
onclick="removeCart(${i})">
Remove
</button>

</div>
`;

}).join("");

$("cartTotal").textContent=
"Total: "+
money(
total,
cartData[0]?.currency||"RWF"
);

openM("cartModal");

}

function removeCart(i){

cartData.splice(i,1);

updateCounts();

cartOpen();

}

async function checkout(){

if(
!loginRequired()||
!cartData.length
)return;

try{

for(let p of cartData){

await api(
"/orders",
{
method:"POST",
body:JSON.stringify({
product_id:p.id,
quantity:1
})
}
);

}

cartData=[];

updateCounts();

cartOpen();

alert(
"✅ Order placed successfully."
);

}catch(e){

alert(
"Order failed: "+e.message
);

}

}

function adminTable(section,rows){

${esc(

section.slice(1)
)}

</div>

`;

}

let keys=[
...new Set(
rows.flatMap(

)
)
].slice(0,12);

$("adminContent").innerHTML=`

<h2>
${esc(

)}

</div>

</div>

<div class="table">

${keys.map(k=>
`<th>${esc(k)}</th>`
).join("")}

</tr>

${rows.map(r=>`

<tr>

${keys.map(k=>`

typeof r?.[k]==="object"

r?.[k]??""
)}

</tr>

`).join("")}

</table>

</div>

`;

}
